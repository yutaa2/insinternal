MD5.java

import java.security.*;

public class MD5 {
    public static void main(String[] args) throws Exception {
        for (byte b : MessageDigest.getInstance("MD5").digest("abc".getBytes()))
            System.out.printf("%02X", b);
    }
}

INPUT (hardcoded): abc

OUTPUT:
900150983CD24FB0D6963F7D28E17F72
